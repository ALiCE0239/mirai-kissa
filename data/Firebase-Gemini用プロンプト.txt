■ Firestore セキュリティルール（このまま貼り付け）
--------------------------------------------------------------------------------
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    match /publicProfiles/{publicId} {
      allow read: if true;
      allow write: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
    }
    match /linkHubs/{publicId} {
      allow read: if true;
      allow write: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
    }
    match /posts/{postId} {
      allow read: if true;
      allow create: if request.auth != null
        && request.resource.data.authorUid == request.auth.uid;
      allow update, delete: if request.auth != null
        && resource.data.authorUid == request.auth.uid;
    }
    match /posts/{postId}/likes/{uid} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == uid;
    }
    match /boardEventAds/{authorUid} {
      allow read: if true;
      allow create, update: if request.auth != null
        && request.auth.uid == authorUid
        && request.resource.data.authorUid == authorUid;
      allow delete: if request.auth != null && request.auth.uid == authorUid;
    }
    match /boardMysekai/{authorUid} {
      allow read: if true;
      allow create, update: if request.auth != null
        && request.auth.uid == authorUid
        && request.resource.data.authorUid == authorUid;
      allow delete: if request.auth != null && request.auth.uid == authorUid;
      match /updates/{updateId} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == authorUid;
      }
      match /likes/{likerUid} {
        allow read: if true;
        allow write: if request.auth != null && request.auth.uid == likerUid;
      }
    }
    match /users/{userId}/friends/{friendUid} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if request.auth != null
        && (request.auth.uid == userId || request.auth.uid == friendUid);
    }
    match /users/{userId}/friendRequests/{requestId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null;
      allow update: if request.auth != null && request.auth.uid == userId;
    }
  }
}

■ Storage セキュリティルール（このまま貼り付け）
--------------------------------------------------------------------------------
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /board/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /users/{userId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    match /users/{userId}/eventArchives/{fileName} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId
        && request.resource.size < 8 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
    }
  }
}

■ 使う Firestore コレクション（データ設計）
--------------------------------------------------------------------------------
linkHubs/{publicId}
  … 公開マイページ（誰でも閲覧、本人のみ編集）
  フィールド例: uid, publicId, displayName, headline, bio, links[], theme, updatedAt

users/{uid}/sns/linkHub
  … 本人用ミラー（linkHubs と同内容）

users/{uid}/sns/auth
  … Xログイン後の xHandle 保存

boardEventAds/{authorUid}
  … イベラン広告（1ユーザー1件、documentId = uid）
  フィールド例: authorUid, authorName, eventName, body, conditionTags[], targetRank,
                 discordURL, imageURL, isPublished, createdAt, updatedAt

boardMysekai/{authorUid}
  … マイセカイ宣伝（1ユーザー1件）
  フィールド例: authorUid, title, body, imageURLs[], likeCount, isPublished, updatedAt
  サブコレクション: likes/{likerUid}

■ Xログインだけ別途必要な作業（Console外・私がやる）
--------------------------------------------------------------------------------
X開発者ポータル (https://developer.x.com/) で:
  1. OAuth 1.0a を有効化
  2. Callback URL に https://cafe-9d3b7.firebaseapp.com/__/auth/handler を登録
  3. API Key / API Secret を Firebase Console の Twitter 設定に貼る
